用React实现一个倒计时组件。
![倒计时](https://blog-1385521233.cos.ap-guangzhou.myqcloud.com/docs/job/倒计时.gif)
## 代码实现
```tsx
export default function CountDown() {
  const initialTime = 10
  const [count, setCount] = useState(initialTime)

  useEffect(() => {
    if (count < 0) {
      setCount(10)
    }
    const id = setInterval(() => {
      setCount(prev => prev - 1)
    }, 1000)
    return () => {
      clearInterval(id)
    }
  }, [count])

  return (
    <div className='bg-emerald-400 h-40 w-80 border-4 shadow-xl border-emerald-700 rounded-2xl flex items-center justify-center px-6 py-4'>
        <div className='font-bold text-3xl text-white'>{count}</div>
    </div>
  )
}
```
