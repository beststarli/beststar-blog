## 代码实现
### 事件对象桶
```js
class EventCenter {
  // 1.定义时间容器，用来装事件数组
  let handlers = {}
  // 2.添加事件方法，参数：事件名，事件方法
  addEventListener(type, handler) {
    // 创建新数组容器
    if (!this.handlers[type]) {
      this.handlers[type] = []
    }
    // 存入事件
    this.handlers[type].push(handler)
  }

  // 3.触发事件，参数：事件名，事件参数
  dispatchEvent(type, params) {
    // 若没有注册该事件则抛出错误
    if (!this.handlers[type]) {
      throw new Error('该事件未注册')
    }
    // 触发事件
    this.handlers[type].forEach((handler) => {
      handler(...params)
    })
  }

  // 4.事件移除，参数：事件名
  // 要删除事件，若无第二个参数则删除该事件的订阅和发布
  removeEventListener(type, handler) {
    if (!this.handlers[type]) {
      throw new Error('该事件未注册')
    }
    if (!handler) {
      // 移除事件
      delete this.handlers[type]
    } else {
      const index = this.handlers[type].findIndex(el => el === handler)
      if (index === -1) {
        return new Error('该事件未注册')
      }
      // 移除事件
      this.handlers[type].splice(index, 1)
      if (this.handlers[type].length === 0) {
        delete this.handlers[type]
      }
    }
  }
}
```

### 事件哈希桶
相对来说用哈希桶实现更好理解一些，下面是一个简单的实现：

| 方法 | 作用 | 类比 |
|:----|:----|:-----|
| `on` | 注册一个监听器（订阅事件） | 把手机号登记给工作人员，有消息就通知你 |
| `off` | 移除一个监听器（取消订阅） | 告诉工作人员不再通知你了 |
| `emit` | 触发一个事件（发布消息） | 工作人员用喇叭对所有登记的人喊一遍 |
| `once` | 注册一个只执行一次的监听器 | 告诉工作人员"只通知我一次，之后别打了" |

```js
class EventEmitter {
  constructor() {
    this.eventBucket = new Map()
  }

  on(event, fn) {
    if (!this.eventBucket.has(event)) {
      this.eventBucket.set(event, [])
    }
    this.eventBucket.get(event).push(fn)
  }

  off(event, fn) {
    if (!this.eventBucket.has(event)) {
      return
    }
    const fnArr = this.eventBucket.get(event) 
    const fnIndex = fnArr.indexOf(fn)
    if (fnIndex !== -1) {
      fnArr.splice(fnIndex, 1)
    }
  }

  emit(event, ...args) {
    const fnArr = this.eventBucket.get(event)
    if (!fnArr) {
      return
    }
    fnArr.forEach(fn => fn(...args))
  }

  once(event, fn) {
    const wrapper = (...args) => {
      fn(...args)
      this.off(event, wrapper)
    }
    this.on(event, wrapper)
  }
}
```
使用示例：
```js
function summarizeEmitterDemo(scenario) {
  const bus = new EventEmitter();
  if (scenario === 'basic') {
    const log = [];
    const handler = () => log.push('tick');
    bus.on('ui', handler);
    bus.emit('ui');
    bus.off('ui', handler);
    bus.emit('ui');
    return log.join(',');
  }
  if (scenario === 'once') {
    let hits = 0;
    bus.once('ready', () => hits++);
    bus.emit('ready');
    bus.emit('ready');
    return hits;
  }
  if (scenario === 'payload') {
    const log = [];
    bus.on('msg', (data) => log.push(data));
    bus.emit('msg', 'hello');
    return log[0];
  }
}
```
预期输出：
```txt
'basic' => 'tick'
'once' => 1
'payload' => 'hello'
```
