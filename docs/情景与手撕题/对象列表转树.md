将对象列表转为树状对象
## 输入输出

示例1:
**输入**：
```txt
[
  {
    "id": 1,
    "name": "部门A",
    "parentId": null
  },
  {
    "id": 2,
    "name": "部门B",
    "parentId": 1
  },
  {
    "id": 3,
    "name": "部门C",
    "parentId": 1
  },
  {
    "id": 4,
    "name": "部门D",
    "parentId": 2
  }
]
```
**输出**：
```txt
[
  {
    "id": 1,
    "name": "部门A",
    "parentId": null,
    "children": [
      {
        "id": 2,
        "name": "部门B",
        "parentId": 1,
        "children": [
          {
            "id": 4,
            "name": "部门D",
            "parentId": 2,
            "children": []
          }
        ]
      },
      {
        "id": 3,
        "name": "部门C",
        "parentId": 1,
        "children": []
      }
    ]
  }
]
```

## 代码实现
```js
function listToTree(items) {
  const map = {}
  items.forEach(item => {
    map.set(item.id, {
      ...item,
      children: []
    })
  })

  const root = []
  items.forEach(item => {
    const node = map.get(item.id)
    const parentId = item.parentId
    if (parentId === null || parentId === 0) {
      root.push(node)
    } else {
      const parentNode = map.get(parentId)
      if (parentNode) {
        parentNode.children.push(node)
      }
    }
  })

  return root
}
```
