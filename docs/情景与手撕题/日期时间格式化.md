将Data对象格式化为指定数字显示格式的时间字符串。
## 代码实现
```js
function formatDate(date, format) {
  const instance = date instanceof Date ? date : new Date(date);
  const pad = (value) => {
    return String(value).padStart(2, '0');
  }

  const year = String(instance.getFullYear());
  const month = pad(instance.getMonth() + 1);
  const day = pad(instance.getDate());
  const hour = pad(instance.getHours());
  const minute = pad(instance.getMinutes());
  const second = pad(instance.getSeconds());

  return format
    .replace('yyyy', year)
    .replace('MM', month)
    .replace('dd', day)
    .replace('HH', hour)
    .replace('mm', minute)
    .replace('ss', second);
}
```
使用示例：
```js
function exerciseFormatDate(label) {
  const sample = new Date(2025, 0, 8, 9, 5, 3);
  if (label === 'ymd') return formatDate(sample, 'YYYY-MM-DD');
  if (label === 'full') return formatDate(sample, 'YYYY/MM/DD HH:mm:ss');
  if (label === 'cnMonthDay') return formatDate(sample, 'MM月DD日');
  if (label === 'midnight') return formatDate(new Date(2025, 5, 3, 0, 7, 9), 'HH:mm:ss');
}
```
输出结果：
```txt
ymd: '2025-01-08'
full: '2025/01/08 09:05:03'
cnMonthDay: '01月08日'
midnight: '00:07:09'
```
