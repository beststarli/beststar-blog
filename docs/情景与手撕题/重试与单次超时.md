实现 retryRequest(fn, times, delayMs, timeoutMs)：fn 无参并返回 Promise；失败或单次超时时按 delayMs 间隔重试，最多 times 次尝试。

注意：全部失败时拒绝最后一次错误；任一次在 timeoutMs 内成功则兑现该结果。
## 代码实现
### 写法一
```ts
async function retryRequest(fn, times, delayMs, timeoutMs) {
  let lastError
  for (let attempt = 0; attempt < times; attempt++) {
    try {
      return await Promise.race([
        fn(),
        new Promise((_, reject) => {
          setTimeout(() => reject(new Error('timeout')), timeoutMs)
        })
      ])
    } catch (error) {
      lastError = error
      if (attempt < times - 1) {
        await new Promise(resolve => {
          setTimeout(resolve, delayMs)
        })
      }
    }
  }
  return lastError
}
```

### 写法二
```ts
async function retryRequest(fn, times, delayMs, timeoutMs) {
  // TODO：循环尝试 + Promise.race 做超时
  const request = () => {
    return Promise.race([
      fn(),
      new Promise((_, reject) => {
        setTimeout(() => {
          reject(new Error('timeout'))
        }, timeoutMs)
      })
    ])
  }

  return new Promise((resolve, reject) => {
    let attempts = 0
    const exec = () => {
      request()
        .then(resolve)
        .catch(err => {
          attempts++
          if (attempts >= times) {
            reject(err)
          } else {
            setTimeout(exec, delayMs)
          }
        })
    }
    exec()
  })
}
```

### 使用示例
```ts
async function runRetryScenario(scenario) {
  if (scenario === 'recovers') {
    let attempts = 0;
    const unstable = () => {
      attempts += 1;
      if (attempts < 3) return Promise.reject(new Error('temp'));
      return Promise.resolve('ok');
    };
    return retryRequest(unstable, 5, 0, 5000);
  }
  if (scenario === 'timeout') {
    const slow = () => new Promise((resolve) => setTimeout(() => resolve('late'), 120));
    try {
      await retryRequest(slow, 2, 0, 25);
      return 'should-have-failed';
    } catch (error) {
      return error instanceof Error ? error.message : 'wrong';
    }
  }
}
```
