返回小写类型标签字符串。需区分 null、array、date、regexp 等与 typeof object 易混类型。
## 代码实现
### 写法一
```ts
function getType(value) {
  // TODO：先处理 null，再 typeof，再细分 object
  if (value === null) {
    return 'null'
  }
  if (value === undefined) {
    return 'undefined'
  }
  const primitive = typeof value
  if (primitive !== 'object') {
    return primitive
  }
  if (Array.isArray(value)) {
    return 'array'
  }
  if (value instanceof Date) {
    return 'date'
  }
  if (value instanceof RegExp) {
    return 'regexp'
  }
  return 'object'
}
```

### 写法二
Object.prototype.toString.call(value) 方法。
```ts
function getType(value) {
  // 先看 typeof 是否直接能定：基本类型都能区分
  const primitive = typeof value;
  if (primitive !== 'object' || value === null) return primitive === 'object' ? 'null' : primitive;
  // ↑ 其实 toString 已经全包了，这行可选
  return Object.prototype.toString.call(value).slice(8, -1).toLowerCase();
}
```

### 使用示例
```ts
function runGetTypeSample(kind) {
  if (kind === 'null') return getType(null);
  if (kind === 'undefined') return getType(undefined);
  if (kind === 'array') return getType([]);
  if (kind === 'object') return getType({ x: 1 });
  if (kind === 'date') return getType(new Date(0));
  if (kind === 'regexp') return getType(/demo/);
  if (kind === 'function') return getType(function () {});
  if (kind === 'number') return getType(3.14);
}
```
