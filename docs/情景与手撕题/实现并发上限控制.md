实现 createConcurrency(limit)，返回函数 run(task)；task 为返回 Promise 的无参函数。任意时刻最多 limit 个 task 同时执行；调用顺序先到先得。
## 代码实现
```ts
function createConcurrency(limit) {
  // TODO：维护 active 计数与等待队列
  let active = 0
  let waiting = []
  async function run(task) {
    while (active >= limit) {
      await new Promise((resolve) => waiting.push(resolve))
    }
    active++
    try {
      return await task()
    } finally {
      active--
      const nextTask = waiting.shift()
      if (nextTask) {
        nextTask()
      }
    }
  }
  return run
}
```
使用示例：
```js
async function measureConcurrencyPeak(limit) {
  const run = createConcurrency(limit);
  let running = 0;
  let peak = 0;
  const job = () =>
    run(
      () =>
        new Promise((resolve) => {
          running += 1;
          peak = Math.max(peak, running);
          setTimeout(() => {
            running -= 1;
            resolve();
          }, 15);
        }),
    );
  await Promise.all([job(), job(), job(), job(), job()]);
  return peak;
}
```
