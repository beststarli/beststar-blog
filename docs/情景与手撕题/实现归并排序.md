通过将问题分解为更小的子问题，递归地解决这些子问题，然后将它们合并起来得到最终的结果。

## 代码实现
```js
function mergeSort(nums) {
  if (nums.length <= 1) {
    return nums
  }

  const mid = Math.floor(nums.length / 2)
  const left = mergeSort(nums.slice(0, mid))
  const right = mergeSort(nums.slice(mid))

  let i = 0
  let j = 0
  const sorted = []
  while (i < left.length && j < right.length) {
    if (left[i] <= right[j]) {
      sorted.push(left[i])
      i++
    } else {
      sorted.push(right[j])
      j++
    }
  }
  while (i < left.length) {
    sorted.push(left[i])
    i++
  }
  while (j < right.length) {
    sorted.push(right[j])
    j++
  }
  return sorted
}
```
