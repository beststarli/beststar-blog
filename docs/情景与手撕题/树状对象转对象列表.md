将树状对象转为对象列表
## 输入输出

示例1:
**输入**：
```txt
{
  "id": 1,
  "name": "Root",
  "children": [
    {
      "id": 2,
      "name": "Child 1",
      "children": [
        {
          "id": 4,
          "name": "Grandchild 1"
        }
      ]
    },
    {
      "id": 3,
      "name": "Child 2"
    }
  ]
}
```
**输出**：
```txt
[
  {
    "id": 1,
    "name": "Root",
    "parentId": null
  },
  {
    "id": 2,
    "name": "Child 1",
    "parentId": 1
  },
  {
    "id": 4,
    "name": "Grandchild 1",
    "parentId": 2
  },
  {
    "id": 3,
    "name": "Child 2",
    "parentId": 1
  }
]
```

示例2:
**输入**：
```txt
{
  "id": 9,
  "name": "Only",
  "children": []
}
```
**输出**：
```txt
[
  {
    "id": 9,
    "name": "Only",
    "parentId": null
  }
]
```

## 代码实现
```js
function treeToList(tree) {
  const list = []
  function dfs(node, parentId) {
    list.push({
      id: node.id,
      name: node.name,
      parentId: parentId
    })
    const kids = node.children || []
    kids.forEach(kid => dfs(kid, node.id))
  }
  dfs(tree, null)
  return list
}
```
