## 解题思路
### 1.DFS
从下往上计算，计算每个节点的深度，判断左右子树的深度差是否大于1，如果大于1则返回false，否则返回当前节点的深度。

- 时间复杂度：O(n)，其中 n 是二叉树的节点数。每个节点恰好被访问一次。
- 空间复杂度：O(n)，其中 n 是二叉树的节点数。

```ts
/**
 * Definition for a binary tree node.
 * class TreeNode {
 *     val: number
 *     left: TreeNode | null
 *     right: TreeNode | null
 *     constructor(val?: number, left?: TreeNode | null, right?: TreeNode | null) {
 *         this.val = (val===undefined ? 0 : val)
 *         this.left = (left===undefined ? null : left)
 *         this.right = (right===undefined ? null : right)
 *     }
 * }
 */

function isBalanced(root: TreeNode | null): boolean {
    function dfs(node) {
        if (node === null) {
            return 0
        }
        const leftDepth = dfs(node.left)
        if (leftDepth === false) {
            return false
        }
        const rightDepth = dfs(node.right)
        if (rightDepth === false) {
            return false
        }

        if (Math.abs(rightDepth - leftDepth) > 1) {
            return false
        }

        return 1 + Math.max(leftDepth, rightDepth)
    }
    return dfs(root) !== false
};
```

### 2.DFS标记
数值与布尔值的结合判断逻辑容易混淆，可以使用一个标记变量来判断是否平衡，避免返回值的混淆。使用一个常量 UNBALANCED 来标记不平衡的情况，返回值为 UNBALANCED 时表示不平衡，否则返回当前节点的深度。

- 时间复杂度：O(n)，其中 n 是二叉树的节点数。每个节点恰好被访问一次。
- 空间复杂度：O(n)，其中 n 是二叉树的节点数。

```ts
/**
 * Definition for a binary tree node.
 * class TreeNode {
 *     val: number
 *     left: TreeNode | null
 *     right: TreeNode | null
 *     constructor(val?: number, left?: TreeNode | null, right?: TreeNode | null) {
 *         this.val = (val===undefined ? 0 : val)
 *         this.left = (left===undefined ? null : left)
 *         this.right = (right===undefined ? null : right)
 *     }
 * }
 */
const UNBALANCED = -1

function isBalanced(root: TreeNode | null): boolean {
    function dfs(node) {
        if (node === null) {
            return 0
        }
        const leftDepth = dfs(node.left)
        if (leftDepth === UNBALANCED) {
            return UNBALANCED
        }

        const rightDepth = dfs(node.right)
        if (rightDepth === UNBALANCED) {
            return UNBALANCED
        }

        if (Math.abs(rightDepth - leftDepth) > 1) {
            return UNBALANCED
        }

        return 1 + Math.max(leftDepth, rightDepth)
    }
    return dfs(root) !== UNBALANCED
};
```
